Quick Start

// Initialize the SDK
Fuul.init({ apiKey: 'your-api-key-here', debug: true });

// Send a pageview
Fuul.sendPageview();

// Send a custom event
Fuul.sendEvent('test_event', { key: 'value' });

// Identify a user
Fuul.identifyUser({
    identifier: '0x123...',
    identifierType: 'evm_address'
});

Available Methods

  • Fuul.init({ apiKey, debug? })
  • Fuul.sendPageview(pageName?, projectIds?)
  • Fuul.sendEvent(name, args?)
  • Fuul.identifyUser({ identifier, identifierType, signature?, message? })
  • Fuul.generateTrackingLink(params)
  • Fuul.createAffiliateCode(params)
  • Fuul.getAffiliateCode(userIdentifier)
  • Fuul.getUserReferrer(params)
  • Fuul.getPayoutsLeaderboard()
  • Fuul.getPointsLeaderboard()
